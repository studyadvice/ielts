data-cell-gap格子間距

使用方式
方法 A HTML 標籤（v1.1 新功能） data-inline → 行內嵌入 
<!-- 行內嵌入：元素保留在文件流，不蓋住頁面內容 -->
<practice-tracker
  data-inline
  data-prefix="ex-"
  data-title="練習進度"
  data-cells-per-row="3"
  data-panel-width="100%"
  data-complete-text="全部完成！"
  data-complete-sub="繼續保持 ✓"
></practice-tracker>
<script src="practice-tracker.js"></script>

<!-- 浮動模式：移除 data-inline，套用 data-position -->
<practice-tracker
  data-position="bottom-right"
  data-prefix="ex-"
  data-title="進度"
></practice-tracker>
方法 B 純 JS 全域配置（原有方式） 不放 HTML 標籤，自動建立浮動面板 
// 在 practice-tracker.js 載入前設定（選用）
window.PracticeTrackerConfig = {
  prefix:       "ex-",
  position:     "bottom-right",
  title:        "進度",
  cellsPerRow:  5,
  completeText: "全部完成",
};
// 未放 <practice-tracker> → 元件自動建立浮動面板
<script src="practice-tracker.js"></script>

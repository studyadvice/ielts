使用方式
<!-- link = id 純字串（不帶 #）。source/target 使用 CSS 選擇器 --> <!-- 單一來源 --> <div id="my-btn">點我</div> <btn-switch link="my-btn" source="#info" target="#output"></btn-switch> <!-- 多來源輪換（逗號分隔，末尾自動回第一個）--> <div id="cycle-btn">切換 ▶</div> <btn-switch link="cycle-btn" source="#s1,#s2,#s3" target="#output"></btn-switch> <!-- 任意元素：id 容器、ui-badge、ui-ring、自訂元件皆可 --> <ui-ring id="ring-btn" center="點我" ...></ui-ring> <btn-switch link="ring-btn" source="#info" target="#output"></btn-switch> <!-- source 不需手動 display:none，元件自動處理 --> <div id="info">任意 HTML 內容</div> <!-- link 缺少 → console.error；找不到 id → console.error -->

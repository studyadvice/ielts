6全域設定 ArrowFlow.config()

在元素初始化前呼叫 ArrowFlow.config() 可設定全站預設值，接受 camelCase 或 kebab-case 鍵名

// 頁面載入後、<arrow-flow> 初始化前呼叫 ArrowFlow.config({ theme : 'ocean', arrowTip : 'xl', // or 'arrow-tip' gap : '88px', animation : 'slide', reveal : 'cumulative', }); // 下方元素無需任何屬性即套用以上設定
此元素未設定任何屬性，色彩與箭頭樣式全部來自 ArrowFlow.config()。 config 中設定的 theme、arrowTip、 animation 全數套用。 若在 HTML 上加了個別屬性，該屬性優先於全域 config。

9API 速查 

<!-- ─── <arrow-flow> 屬性 ──────────────────────────────────────────────── -->
<arrow-flow
  theme="sky"             <!-- 全部框統一品牌色 (預設: shell)               -->
  header-color="sky"    <!-- 覆蓋所有框 header 底色（優先於 theme）       -->
  border-color="sky"    <!-- 覆蓋所有框線色                               -->
  arrow-color="shell"  <!-- 箭頭填色（預設繼承 theme）                   -->
  arrow-tip="md"       <!-- sm / md / lg / xl（只改頭部三角形）          -->
  arrow-style="solid"  <!-- solid（預設實線）/ dashed（虛線桿＋實心頭）  -->
  gap="72px"           <!-- 框間距（整個可點擊箭頭區域的寬度）           -->
  reveal="cumulative" <!-- cumulative（前框保留）/ single（前框淡出）   -->
  animation="fade"    <!-- fade / slide / none                         -->
  equal-height          <!-- Boolean：強制所有框等高（取最大值）           -->
  current="0"          <!-- 初始顯示至第幾框（0-based）                  -->
  on-complete="div-id" <!-- 完成後顯示的 div id（初始自動隱藏）          -->
  step-width="240px"   <!-- 所有框固定寬度（省略 = flex:1 均分）          -->
>
  <flow-step
    title="標題"
    header-color="lavender"  <!-- 覆蓋此框 header 底色 -->
    border-color="indigo"    <!-- 覆蓋此框框線色       -->
    width="300px"           <!-- 覆蓋此框寬度（優先於 step-width） -->
  >
    支援 <strong>HTML</strong> 標籤、<ul> 列表等
  </flow-step>
</arrow-flow>

/* ─── 全域設定 ───────────────────────────────────────────────────────── */
ArrowFlow.config({
  theme      : 'sky',
  arrowTip   : 'lg',    // 同 arrow-tip 屬性，接受 camelCase 或 kebab-case
  gap        : '80px',
  animation  : 'slide',
  reveal     : 'cumulative',
  equalHeight: true,
  stepWidth  : '260px', // 同 step-width 屬性
  arrowStyle : 'dashed',// 同 arrow-style 屬性
});

① 最基本用法 — 純 HTML，零 JavaScript
← 點擊上方日期

<week-cal

  locale="zh-TW"

  target="#board-a"

  show-header

>
  <wc-day
    date="today"
    source-id="da1"
    label="今天"
  ></wc-day>
  <wc-day

    source-id="da2"
  ></wc-day>
</week-cal>
② 監聽 wc-select 事件（bubbles: true，可在父元素捕捉）
<week-cal
  id="cal-b"
  theme="ocean"
  locale="en"
>…</week-cal>

// 監聽自訂元素發出的事件
document
  .querySelector('#cal-b')
  .addEventListener('wc-select', e => {
    console.log(e.detail.date);
    console.log(e.detail.sourceId);
  });
③ 元素方法 .select() / .getConfig() / .refresh()
← 點擊日期或按鈕
// 程式化選取
document
  .querySelector('#cal-c')
  .select('2025-09-20');

// 取得目前設定
const cfg = document
  .querySelector('#cal-c')
  .getConfig();
console.log(cfg.theme); // 'indigo'

// 動態新增子元素後呼叫
document
  .querySelector('#cal-c')
  .refresh();
④ 動態改變屬性 → attributeChangedCallback 自動重新掛載
// 屬性一改，自動重新掛載
document
  .querySelector('#cal-d')
  .setAttribute('theme', 'aurora');

// 切換語系
document
  .querySelector('#cal-d')
  .setAttribute('locale', 'en');

// attributeChangedCallback 內部流程：
// 1. WeekCal.destroy(this)  — 清除舊實例
// 2. WeekCal.init(newOpts)  — 以新屬性重新掛載
⑤ 全主題一覽（v1 保留 6 個 + v2.0 新增 8 個，含漸層主題 aurora）
aurora 漸層主題說明：CSS 自訂屬性（Custom Properties）的值可以是任意合法 CSS 片段，因此 calBg、todayBg、selectedBg 三個欄位直接填入 linear-gradient(…) 字串即可。這些變數對應的 CSS 規則皆為 background: var(--wc-*)，瀏覽器會正確套用漸層。文字色（shell、 todayText 等）因為對應 color 屬性，仍須使用純色值。 要自訂漸層主題，只需：
WeekCal.themes.myGrad = { calBg: 'linear-gradient(…)', todayBg: 'linear-gradient(…)', … }
⑥ icon 模式 — Bootstrap Icons + 自訂標籤文字（需網路連線載入 CDN）
設定 mode="icon" 後，元件會自動從 cdnjs 注入 Bootstrap Icons CSS（每頁只注入一次）。 <wc-day> 的 icon 屬性接受 Bootstrap Icons 名稱， 寫 calendar-check 或 bi-calendar-check 均可； label 屬性取代日期數字，顯示在圖示下方。 disabled 布林屬性讓格子呈現半透明、不可點擊。
← 點擊圖示格
<!-- mode="icon" 觸發 Bootstrap Icons 自動注入 -->
<week-cal
  mode="icon"
  theme="aurora"
  locale="zh-TW"
  target="#board-icon"
  show-header
  highlight-today
  default-first
>
  <wc-day date="today"
    source-id="di1"
    icon="calendar-check"
    label="今天"></wc-day>
  <wc-day date="tomorrow"
    source-id="di2"
    icon="rocket-takeoff"
    label="明天"></wc-day>
  <wc-day date="next Mon"
    source-id="di4"
    icon="star"
    label="下週一"></wc-day>
  <wc-day date="next Tue"
    icon="slash-circle"
    label="封鎖"
    disabled></wc-day>
</week-cal>

// 命令式等效寫法（v1 API 完全保留）
WeekCal.init({
  container: '#cal',
  mode:      'icon',
  theme:     'aurora',
  days: [
    { date: 'today',    sourceId: 'di1',
      icon: 'calendar-check', label: '今天' },
    { date: 'next Mon', sourceId: 'di4',
      icon: 'star',           label: '下週一' },
    { date: 'next Tue',
      icon: 'slash-circle',   label: '封鎖',
      disabled: true },
  ],
});
